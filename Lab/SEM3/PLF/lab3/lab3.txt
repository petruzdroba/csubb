6. Sa se genereze toate sirurile de n paranteze ce se inchid corect. Exemplu:
n=4 sunt 2 solutii: (()) si ()()

paranteze(open, close, rez) ={
    [rez]                                     if open = 0 and close = 0
    paranteze(open - 1, close, rez ++ ['('] )   if open > 0
    paranteze(open, close - 1, rez ++ [')'] )   if close > open
}

main_paranteze(N) = {
    paranteze(N/2, N/2, "")
}

% convert a list of parentheses chars to a string
list_to_string(L, S) :-
    atomic_list_concat(L, '', S).

% print all sequences in a pretty way
print_paranteze(N) :-
    main_paranteze(N, R),
    forall(
        member(X, R),
        (
            list_to_string(X, S),
            write(S), nl
        )
    ).
